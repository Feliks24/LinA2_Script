\section{Einführung}

\textit{\small \hspace{-1.5cm}Woche 2}

\textbf{Definition:} Eine Menge $R$ zusammen mit zwei Verknüpfungen
\begin{align*}
	&+:R\times R\to R, (a,b)\mapsto a+b\\
	&\cdot:R\times R \to R, (a,b)\mapsto a\cdot b
\end{align*}
heißt \textbf{Ring}, wenn folgendes gilt:

\begin{itemize}
\item[\textbf{R1}] $R$ zusammen mit der Addition $+$ ist eine abelsche Gruppe
\item[\textbf{R2}] Die Multiplikation $\cdot$ ist assoziativ
\item[\textbf{R3}] Es gelten das Distributivgesetz
\end{itemize}

\textbf{Definition:} Ein Ring $R$ ist ein Integritätsring, falls gilt:
\begin{itemize}
\item[$i)$] $R$ ist kommutativ mit Einselement $1\neq0$
\item[$ii)$] $R$ ist nullteilerfrei
\end{itemize}

\textbf{Anmerkung:} Ein Nullteiler für $a\neq0$ sei ein $b\neq0$, sodass
$ab=0$. Beispiel:
\begin{equation*}
\begin{pmatrix}
	1 & 0\\
	0 & 0
\end{pmatrix}
\begin{pmatrix}
	0 & 0\\
	0 & 1
\end{pmatrix}=0
\end{equation*}

\textbf{Definition:} Ein Ring $R$ heißt \textbf{nullteilerfrei}, wenn für
alle $a,b\in R$ aus $a\cdot b=0$ stets $a=0$ oder $b=0$ folgt.

\textbf{Definition:} Ist $R$ ein Ring und $R'\subset R$ eine Teilmenge, so
heißt $R'$ \textbf{Unterring}, wenn $R'$ bezüglich der Addition Untergruppe
ist, und bezüglich der Multiplikation für $a,b\in R'$ auch $a\cdot b\in R'$.

Sind $R$ und $S$ Ringe mit Verknüpfungen $+,\cdot$ und $\oplus,\odot$, so heißt
eine Abbildung $\varphi: R\to S$ ein \textbf{Homomorphismus von Ringen}, wenn
für alle $a,b\in R$ gilt:
\begin{align*}
	\varphi(a+b) = \varphi(a)\oplus\varphi(b)\\
	\varphi(a\cdot) = \varphi(a)\odot\varphi(b)
\end{align*}
Zum Beispiel ist $m\mathbb{Z}\subset\mathbb{Z}$ ein Unterring und
$\mathbb{Z}\to\mathbb{Z}/m\mathbb{Z}, a\mapsto a+m\mathbb{Z}$, ein
Homomorphismus.

\textbf{Definition:} Ist $R$ ein Ring mit Einselement $1\neq 0$, so ist seine
\textbf{Charakteristik} erklärt durch
\begin{align*}
	char(R)&:= 0 \qquad \text{falls } n\cdot 1\neq 0 \text{ für alle } n\geq
	1\\
	&:=min\{n\in\mathbb{N}\backslash {0}:n\cdot 1 = 0\} \qquad \text{sonst}
\end{align*}
\textbf{Lemma:} Ist $K$ Körper, so ist char(K) entweder Null oder eine Primzahl

\textbf{Wiederholung} Komplexe Zahlen: 
\begin{itemize}
\item $\mathbb{C}=\{x+iy: x,y\in\mathbb{R}\}$ wobei $i=\sqrt{-1}$
\item $\mathbb{C}$ ist ein Körper mit Charakteristik 0
\item $|x+iy| = \sqrt{x^2 + y^2}$
\end{itemize}

\textbf{Wiederholung (gekürzt)} Sei $K$ ein Körper, $t$ eine formale Variable
und $a_0,\dots,a_n\in K$. Dann nennt man den Ausdruck 
\begin{equation*}
	f(t)=a_0t^0+a_1t^1+\dots+a_nt^n
\end{equation*}
ein \textbf{Polynom} in $t$. Sei $f(t)\in K[t]:=\{$Polynome in $t$ mit
Koeffizienten as $K\}$

\textbf{Bemerkung:} Ist $K$ ein Körper, so ist die Menge $K[t]$ der Polynome
übre $K$ zusammen mit den oben definierten Verknüpfungen ein kommutativer Ring
ohne Nullteiler. Weiter gilt
\begin{equation*}
	deg(f\cdot g) = deg\ f+ deg\ g \qquad \text{für } f,g\in K[t]
\end{equation*}

Dabei soll formal $n-\infty = -\infty+m= -\infty + (-\infty) = -\infty$ sein

\textbf{Definition: }Sind $a,b\in R$, so heißt $b$ \textbf{Teiler} von $a$
(also $b|a$), wenn es ein $c\in R$ gibt mit $a=cb$. 

Ein Element $a\in R$ heißt eine \textbf{Einheit}, wenn es invertierbar ist,
d.h., wenne es $\tilde{a}\in R$ gibt mit $a\tilde{a}= 1$.

Sind $a,b\in R$ gegeben, so heißt ein $c\in R$ \textbf{gemeinsamer Teiler} von
$a$ und $b$, wenn $c|a$ und $c|b$. 

Ein gemeinsamer Teiler $d$ von $a$ und $b$ heißt ein \textbf{größter
gemeinsamer Teiler(ggt)}, wenn $c|d$ für jeden gemeinsamen Teiler $c$.

\textbf{Satz:} Sind $f,g\in K[t]$, und ist $g\neq 0$, so gibt es dazu eindeutig
bestimmte Polynome $q,r\in K[t]$ derart, dass 
\begin{equation*}
	f=qg+r\text{ und }deg\ r < deg\ g.
\end{equation*}

\textbf{Satz:} In den Ringen $R=\mathbb{Z}$ und $R=K[t]$ gibt es zu Elementen
$a,b\in R$, die nicht beide 0 sind, stes einen größten gemeinsamen Teiler $d\in
R\backslash\{0\}$.

\textbf{Relation von Bézout:} Sei $R=\mathbb{Z}$ oder $R=K[t]$. Dann gibt es zu
$a,b\in R\backslash \{0\}$ Koeffizienten $x,y\in R$ derart, dass $ggT(a,b) = xa
+yb$.

\textbf{Lemma:} Ist $\lambda\in K$ eine Nullstelle von $f\in K[t]$, so gibt es
ein eindeutig bestimmtes Polynom $g\in K[t]$ mit 
\begin{equation*}
	f=(t-\lambda)g\text{ und es gilt } deg\ g=(deg\ f)-1.
\end{equation*}

\textbf{Korollar 1:} Sei $K$ ein beliebiger Körper, $f\in K[t]$ ein Polynom und
$k$ die Anzahl der Nullstellen von $f$. Ist $f$ vom Nullpolynom verschieden, so
gilt $k\leq deg\ f$.

\textbf{Definition:} Ist $f\in K[t]$ vom Nullpolynom verschieden und
$\lambda\in K$, so heißt 
\begin{equation*}
\mu(f;\lambda):=max\{r\in\mathbb{N}: f=(t-\lambda)^rg \text{ mit } g\in K[t]\}
\end{equation*}
die Vielfachheit der Nullstelle $\lambda$ von $f$.

\textbf{Fundamentalsatz der Algebra:} Jedes Polynom $f\in \mathbb{C}[t]$ mit
$deg\ f>0$ hat mindestens eine Nullstelle in $\mathbb{C}$.

\textbf{Korollar:} Jedes Polynom $f\in\mathbb{C}[t]$ zerfällt in
Linearfaktoren, d.h. es gibt $a$ und $\lambda_1, \dots, \lambda_n\in\mathbb{C}$
mit $n=deg\ f$, sodass $f=a(t-\lambda_1)\cdot\dots \cdot (t-\lambda_n)$.

\section{Eigenwerte \& Eigenräume}

\textit{\small \hspace{-1.5cm}Woche 3}

\textbf{Definition:} Sei $F$ ein Endomorphismus des $K$-Vektorraums $V$. Ein
$\lambda\in K$ heißt Eigenwert von $F$, wenn es ein $v\in V$ mit $v\neq0$ gibt,
sodass gilt $F(v) = \lambda v$. 

Jedes vom Nullvektor verschiedene $v\in V$ mit $F(v)=\lambda v$ heißt
Eigenvektor von $F$ (zum Eigenwert $\lambda$).

\textbf{Definition:} Ist $F$ ein Endomorphismus von $V$ und $\lambda\in K$, so
nennen wir $Eig(F;\lambda):=\{v\in V:F(v)=\lambda v\}$ den Eigenraum von $F$
bezüglich $\lambda$.

\textbf{Bemerkung:} 
\begin{itemize}
\item[a)] $Eig(F;\lambda)\subset V$ ist ein Unterektorraum
\item[b)] $\lambda$ ist Eigenwert von $F$ $\Leftrightarrow$
	$Eig(F;\lambda\neq\{0\}$
\item[c)] $Eig(F;\lambda)\backslash\{0\}$ ist die Menge der zu $\lambda$
	gehörigen Eigenvektoren von $F$
\item[d)] $Eig(F;\lambda) = Ker(F-\lambda I_V)$
\item[e)] Sind $\lambda_1,\lambda_2\in K$ verschieden, so ist
	$Eig(F;\lambda_1)\cap Eig(F;\lambda_2)=\{0\}$
\end{itemize}

\textbf{Lemma:} Gegeben seien $F\in End(V)$ und Eigenvektoren $v_1,\dots, v_m$
zu paarweise verschiedenen Eigenwerten $\lambda_1,\dots, \lambda_m$. Dann sind
$v_1,\dots,v_m$ linear unabhängig. Insbesondere ist $m\leq dim\ V$.

\textbf{Definition:} Sei $F$ ein Endomorphismus und $\mathcal{A}$ eine Basis von
$V$. Ist $A=M_\mathcal{A}(F)$, so nnenen wir $P_F(t):=P_A(t)$ das
charakteristische Polynom vom $F$.

\textbf{Satz:} Sei $V$ ein $K$-Vektorraum der Dimension $n<\infty$ und $F$ ein
Endomorphismus von $V$. Dann hat das charakteristische Polynom $P_F(t)\in K[t]$
die folgenden Eigenschaften:

\begin{enumerate}
	\item $deg\ P_F=n$
	\item $P_F$ beschreibt die charakteristische Funktion $\tilde{P}_F:K\to
		K, \lambda\mapsto det(F-\lambda I_V)$
	\item Die Nullstellen von $P_F$ sind die Eigenwerte von F
	\item Ist $A$ eine Matrix, die $F$ darstellt, so ist $P_F(t)=det(A-t E_n)$
\end{enumerate}

\textbf{Satz:} Sei $V$ ein endlichdimensionaler $K$-Vektorraum und $F\in
End(V)$. Dann sind die folgenden Bedingungen äquivalent:

\begin{itemize}
\item[i)] $F$ ist diagonalisierbar
\item[ii)]   	\begin{itemize}
			\item[a)] Das charakteristische Polynom $P_F$ zerfällt
				in Linearfaktoren und
			\item[b)] $dim\ Eig(F;\lambda)=\mu(P_F;\lambda)$ für
				alle Eigenwerte $\lambda$ von $F$, d.h., die
				geometrische Vielfachheit ist gleich der
				algebraischen Vielfachheit
		\end{itemize}
\item[iii)] Sind $\lambda_1,\dots,\lambda_k$ die paarweise verschiedenen
	Eigenwerte von $F$, so ist $V = Eig(F;\lambda_1)\oplus\dots\oplus
	Eig(F;\lambda_k)$
\end{itemize}

\textbf{Definition:} Ein Endomorphismus heißt diagonalisierbar, wenn es eine
Basis aus Eigenvektoren gibt. 
\begin{equation*}
	A=T^{-1} D T
\end{equation*}

\textbf{Bemerkung:} Ist $dim\ V=n<\infty$, so ist $F\in End(V)$ genau dann
diagonalisierbar, wenn es eine Basis $\mathcal{B}=(v_1,\dots,v_n)$ von $V$
gibt, sodass $M_\mathcal{B}(F)$ eine Diagonalmatrix ist, d.h.
\begin{equation*}
	M_\mathcal{B}(F)=
	\begin{pmatrix}
		\lambda_1 & & 0\\
		& \ddots & \\
		0 & & \lambda_n
	\end{pmatrix}
\end{equation*}

\textbf{Definition:} Sei $F:V\to V$ ein Endomorphismus und $\lambda$ ein
Eigenwert von $F$. Dann heißt $\mu(P_F;\lambda)$ die algebraische Vielfachheit
von $\lambda$ und $dim\ Eig(F;\lambda)$ die geometrische Vielfachheit von
$\lambda$.

\textbf{Lemma:} Ist $\lambda$ Eigenwert von $F$, so gilt $1\leq dim\
Eig(F;\lambda)\leq \mu(P_F;\lambda)$.

\textbf{Definition:} Ein Vektorraum $V$ heißt direkte Summe von
Untervektorräumen $W_1,\dots, W_k$, (in Zeichen $V=W_1\oplus\dots\oplus W_k$),
wenn folgende Bedingungen erfüllt sind:

\begin{itemize} 
\item[DS1] $V=W_1+\dots W_k$
\item[DS2] Sind $w_1\in W_1,\dots,w_k\in W_k$ gegeben mit $w_1+\dots w_k=0$, so
	folgt $w_1=\dots=w_k=0$
\end{itemize}

\textbf{Satz:} Für Untervektorräume $W_1,\dots,W_k$ eines endlichdimensionalen
Vektorraums $V$ sind folgende Bedingungen äquivalent:

\begin{itemize}
\item[i)] $V=W_1\oplus\dots\oplus W_k$

\item[ii)] ist für jedes $i\in\{1,\dots,k\}$ eine Basis $(v_1^{(i)},\dots,
	v_{r_i}^{(i)})$ von $W_i$ gegeben, so ist $\mathcal{B}:=(v_{1}^{(1)},
	\dots, v_{r_1}^{(1)},\dots , v_{1}^{(k)}, \dots , v_{r_k}^{(k)})$ eine
	Basis von $V$

\item[iii)] $V=W_1+\dots+W_k$ und $dim\ V=dim\ W_1+ \dots+dim\ W_k$
\end{itemize}

\newpage
\textbf{Satz:} Sei $V$ ein endlichdimensionaler $K$-Vektorraum und $F\in
End(V)$. Dann sind die folgenden Bedingungen äquivalent:

\begin{itemize}
\item[i)] $F$ ist diagonalisierbar
\item[ii)]
	\begin{itemize}
	\item[a)] Das charakteristische Polynom $P_F$ zerfällt in
		Linearfaktoren und
	\item[b)] $dim\ Eig(F;\lambda)=\mu(P_F;\lambda)$ für alle Eigenwerte
		$\lambda$ von $F$
	\end{itemize}
\item[iii)] Sind $\lambda_1,\dots, \lambda_k$ die paarweise verschiedenen
	Eigenwerte von $F$, so ist $V=Eig(F;\lambda_1)\oplus\dots\oplus
	Eig(F;\lambda_k)$
\end{itemize}


\section{Trigonalisierbarkeit}

\textit{\small \hspace{-1.5cm}Woche 4}

\textbf{Definion:} Sei $F\to V$ ein Endomorphismus und $W\subset V$ ein
Untervektorraum. $W$ heißt $F$-invariant, wenn $F(W)\subset W$.

\textbf{Bemerkung:} Ist $W\subset V$ ein $F$-invarianter Unterraum, so ist
$P_{F|W}$ ein Teiler von $P_F$.

\textbf{Definition:} Unter einer Fahne $(V_r)$ in einem $n$-dimensionalen
Vektorraum $V$ versteht man eine Kette 
\begin{equation*}
\{0\} =V_0\subset
V_1\subset\dots\subset V_n=V
\end{equation*}

von Untervektorräumen mit $dim\ V_r=r$. Ist $F\in End(V)$, so heißt die Fahne
$F$-invariant, wenn $F(V_r)\subset V_r$ für alle $r\in\{0,\dots,n\}$

\textbf{Definition:} Ein Endomorphismus heißt trigonalisierbar, wenn es eine
obere Dreiecksmatrix als Darstellungsmatrix gibt. Eine Matrix heißt
trigonalisierbar, wenn sie ähnlich ist zu einer oberen Dreiecksmatrix.

\textbf{Bemerkung:} Für $F\in End(V)$ sind folgende Bedingungen gleichwertig:
\begin{itemize}
\item[i)] Es gibt eine $F$-invariante Fahne in V
\item[ii)] Es gibt eine Basis $\mathcal{B}$ von $V$, sodass $M_\mathcal{B}(F)$
	obere Dreiecksmatrix ist
\item[iii)] $F$ ist trigonalisierbar
\end{itemize}

\textbf{Trigonalisierungssatz:} Für einen Endomorphismus $F$ eines
$n$-dimensionalen $K$-Vektorraums sind folgende Bedingungen äquivalent:
\begin{itemize}
\item[i)] $F$ ist trigonalisierbar
\item[ii)] Das charakteristische Polynom $P_F$ zerfällt in Linearfaktoren, d.h.
	\begin{equation*}
		P_F=\pm(t-\lambda_1)\cdot\dots\cdot(t-\lambda_n)\qquad\text{mit
		}\lambda_1,\dots,\lambda_n\in K
	\end{equation*}
\end{itemize}

\textbf{Korollar:} Jeder Endomorpismus eines endlichdimensionalen komplexen
Vektorraums ist trigonalisierbar.

\section{Jordan'sche Normalform}
\textit{\small \hspace{-1.5cm}Woche 5}

\textbf{Definition:} Ein Jordan-Block der Größe zum Eigenwerte
$\lambda$ ist die quadratische $n\times n$-Matrix
\begin{equation*}
	J_n(\lambda):=
\begin{pmatrix}
	\lambda & 1 & & & & 0\\
	& \lambda & 1 & & &  \\
	& & \ddots & \ddots & &  \\
	& &  & & \lambda & 1 \\
	0 & &  & & & \lambda\\
\end{pmatrix}
\end{equation*}

\textbf{Definition:} Eine Jordan-Matrix ist eine Blockdiagonalmatrix mit
Jordan-Blöcken in der Diagonalen, d.h. eine Matrix der Form
\begin{equation*}
	\begin{pmatrix}
		J_{r_1}(\lambda_1) & & & 0\\
		& J_{r_2}(\lambda_2)  & & \\
		& & \ddots &\\
		0 & & & J_{r_n}(\lambda_n)\\
	\end{pmatrix}
\end{equation*}

wobei die Eigenwerte $\lambda_1,\dots,\lambda_l$ der Jordan-Blöcke nicht
verschieden sein müssen. Man sagt auch, eine Matrix habe Jordansche Normalform,
wenn sie eine Jordan-Matrix ist.

Die Summe der Größen der Jordan-Blöcke zu einem Eigenwerte $\lambda$ ist also
die algebraische Vielfachheit des Eigenwerts:
$\mu(P_F;\lambda)=\sum_{j:\lambda_j=\lambda}r_j.$

Die Anzahl der Jordan-Blöcke zu einem Eigenwert $\lambda$ ist also die
geometrische Vielfachheit des Eigenwerts: $dim\
Eig(F;\lambda)=\sum_{j:\lambda_j=\lambda}1$.

\textbf{Anmerkung:} Somit sind Diagonalmatrizen Spezialfälle von
Jordan-Matrizen, wobei jeder Jordan-Block jeweils die Größe $1\times1$ hat.

\textbf{Satz von der Jordanschen Normalform:} Sei $V$ ein endlichdimensionaler
Vektorraum übre einem algebraisch abgeschlossenen Körper, und sei $F\in
End(V)$. Dann gibt es eine Basis $\mathcal{B}$ von $V$, bezüglich der die
darstellende Matrix $M_\mathcal{B}(F)$ Jordansche Normalform hat. Diese
Jordan-Matrix $M_\mathcal{B}(F)$ ist bis auf die Reihenfolge der Jordan-Blöcke
durch den Endomorphismus $F$ eindeutig bestimmt.

\textbf{Satz von der Jordanschen Normalform (Matrix):} Sei $K$ ein algebraisch
abgeschlossener Körper, und sei $A\in M(n\times x;K)$. Dann ist $A$ ähnlich zu
einer Matrix $J$ in Jordanscher Normalform. Das heißt, es gibt eine Matrix
$T\in GL(n;K)$, sodass $J=TAT^{-1}$ eine Jordan-Matrix ist. Diese Jordan-Matrix
$J$ ist bis auf die Reihenfolge der Jordan-Blöcke durch die Matrix $A$
eindeutig bestimmt.

\textbf{Satz:} Quadratische Matrizen über einem algebraisch abgeschlossenen
Körper sind genau dann ähnlich, wenn sie (bis auf die Reihenfolge der
Jordan-Blöcke) die gleiche Jordansche Normalform haben.

\textbf{Anmerkung:} Ein algebraisch abgeschlossener Köper $K$ ist ein Körper
für welchen gilt: $K[t]$ zerfällt in Linearfaktoren. Zum Beispiel $\mathbb{C}$
aber nicht $\mathbb{R}$

\textbf{Definition:} Gegeben sei $K$-Vektorraum $V$ und $F\in End(V)$ dann sei
\begin{itemize}
\item[i)] ein Hauptvektor erste Stufe zum Eigenwert $\lambda$ der Eigenvektor
	von $F$ zum Eigenwert $\lambda$
\item[ii)] ein Hauptvektor der Stufe $k\geq2$ zum Eigenwert $\lambda$ ein
	Vektor $v\in V$, sodass $F(v)=\lambda v+ v'$ wobei $v'$ ein Hauptvektor
	der Stufe $k-1$ zum Eigenwert $\lambda$ ist
\end{itemize}

\textbf{Lemma:} Ein Vektor $v\in V$ ist genau dann Hauptvektor der Stufe
$k\geq1$ zum Eigenwert $\lambda$, wenn 
\begin{equation*}
(F-\lambda I_V)^k(v)=0\quad\text{aber}\quad(F-\lambda I_V)^{k-1}(v)\neq0.
\end{equation*}

\textbf{Definition:} Eine Familie $(v_1,\dots, v_k)$ heißt eine Jordan-Kette
von $F$ zum Eigenwert $\lambda$, wenn $v_k$ ein Hauptvektor der Stufe $k$ ist
und für $j\in\{1,\dots,k-1\}$ gilt $v_j=(F-\lambda I_V)(v_{j+1})$. Insbesondere
ist dann $v_j$ ein Hauptvektor der Stufe $j$ zum selben Eigenwert $\lambda$.

\textbf{Satz von der Jordanschen Normalform (Jordan-Ketten):} Sei $V$ ein
endlichdimensionaler Vektorraum über einem algebraisch abgeschlossenen Körper,
und sei $F\in End(V)$. Dann gibt es eine Basis $\mathcal{B}$ von $V$, die aus
Jordan-Ketten von $F$ besteht. Dabei sind die Anzahlen der Jordan-Ketten zu
jedem Eigenwert und ihre Längen durch den Endomorphismus $F$ eindeutig
bestimmt.

\textbf{Algorithmus:} Bestimmung der Jordanschen Normalform
\begin{enumerate}
	\item Eigenwerte bestimmen
	\item Für jeden Eigenwert
		\begin{enumerate}
			\item bestimme linear unabhängige Hauptvektoren der
				höchsten Stufe
			\item berechne zugehörige Jordan-Kette
		\end{enumerate}
\end{enumerate}

\textbf{Anmerkung:} Zu jedem Eigenwert $\lambda$ exestiert ein $l$, sodass alle
Hauptvekotren in $Ker\ (F-\lambda I_V)^l$ wobei sich der Kern für $l+1$ nicht
verändert. Zu bemerken ist, dass $l$ dann die Größe des größten Jordan-Blocks
ist zum Eigenwert $\lambda$.

\section{Polynome und Endomorphismen}

\textit{\small \hspace{-1.5cm}Woche 6}

\textbf{Definition:} Sei $p=a_0+a_1t+\dots a_dt^d\in K[t]$ und $F\in End(V)$
dann gilt $p(F):=a_0I_V+a_1F+\dots+a_dF^d\in End(V)$ wobei $F^k=F\circ\dots
\circ F$. Sei weiterhin $A\in M(n\times n,K)$ dann sei $p(A):= a_0 E_n
+a_1A+\dots+a_dA^d\in M(n\times n,K)$.

\textbf{Beobachtungen:} 

a) Sei $v$ Eigenvektor von $F$ zum Eigenwert $\lambda$. So
ist $v$ auch ein Eigenvektor von $p(F)$ zum Eigenwert $p(\lambda)$

Sei $V$ ein endlichdimensionaler $K$-Vektorraum und $F\in End(V)$ sei
diagonalisierbar, dann gilt $p_F(F)=0$. Parallel dazu gilt wenn $A\in M(n\times
n, K)$ diagonalisierbar dann ist auch $p_A(A)=0$.

b) Sei $F$ diagonalisierbar, dann ist auch $p(F)$ diagonalisierbar.

c) Gegeben sei eine Diagonalmatrix $D$ dann gilt
\begin{equation*}
	D=\begin{pmatrix}
	\lambda_1 & & 0\\
	& \ddots &\\
		0& & \lambda_n
	\end{pmatrix}
\implies
D^k=\begin{pmatrix}
	\lambda_1^k & & 0\\
	& \ddots &\\
		0& & \lambda_n^k
	\end{pmatrix}
	\text{ und  }
p(D)=\begin{pmatrix}
	p(\lambda_1) & & 0\\
	& \ddots &\\
		0& & p(\lambda_n)
	\end{pmatrix}
\end{equation*}

d) Sei $A$ und $B=TAT^{-1}$ ähnlich mit $T\in GL(n,K)$, so gilt $p(A)$ und
$p(B)=Tp(A)T^{-1}$ ähnlich.

\textbf{Definition:} Die Einsetzungshomomorpihsmen für Elemente $x\in K,F\in
End(V)$ und $A\in M(n\times n; K)$ sind die Abbildungen
\begin{alignat*}{2}
	&ev_x:\ K[t] \to K, \quad && ev_x(p) = p(x)\\
	&ev_F:\ K[t] \to End(V), && ev_F(p) = p(F)\\
	&ev_A:\ K[t] \to M(n\times n;K), && ev_A(p)=p(A)
\end{alignat*}

\textbf{Bemerkung:} Die Einsetzungshomomorphismen sind Homomorphismen von
Algebren, d.h. sie sind lineare Abbildungen und Homomorphismen von Ringen.

Für Homomorphismen von Ringen gilt
\begin{align*}
	\mu(a+b) = \mu(a) \oplus \mu(b)\\
	\mu(a\times b) = \mu(a) \otimes \mu(b)
\end{align*}

\textbf{Beobachtung:} $Ker(ev_F)=$ die Menge aller Polynome $p\in K[t]$ mit
$p(F)=0$.

\textbf{Beobachtung:} $Ker(ev_F)$ ist ein Ideal von $K[t]$.

\textbf{Wiederholung:} Gegeben sei ein Ring $R$, dann ist $\mathcal{I} \subset
R$ ein Ideal falls gilt:

\begin{itemize}
\item[i)] $\forall x,y\in \mathcal{I}: x+y\in\mathcal{I}$ genauer
	$(\mathcal{I},+)$ ist eine abelsche Gruppe.
\item[ii)] $\forall r\in R\ \forall z\in\mathcal{I}: r\cdot z\in\mathcal{I}$
\end{itemize}

\textbf{Definition:} Für jedes Element $m\in R$ ist $mR=\{m\cdot r: r\in R\}$
ein Ideal, das von m erzeugte Ideal.

\textbf{Satz:} Wenn $\mathcal{I}$ ein Ideal von $\mathbb{Z}$ ist, dann ist
$\mathcal{I}=m\mathbb{Z}$, wobei $m$ entweder die kleinste positive Zahl ist,
di in $\mathcal{I}$ enthalten ist, oder $m=0$, wenn $\mathcal{I}=\{0\}$.

\textbf{Satz:} Jedes Ideal $\mathcal{I}$ von $K[t]$ ist entweder das Nullideal
oder es gibt genau ein normiertes Polynom $m\in K[t]$ mit $\mathcal{I}=mK[t]$.

\newpage
\textbf{Definition:} Für den Endomorphismus $F\in End(V)$ sei das Ideal
$\mathcal{I}_F$ nicht das Nullideal. Dann nennt man das eindeutig bestimmte
normierte Polynom, das das Ideal $\mathcal{I}_F$ erzeugt, das Minimalpolynom
von $F$. Wir bezeichnen das Minimalpolynom von $F$ mit $M_F$. Für eine Matrix
$A\in M(n\times n;K)$ definiert man ganz analog das Ideal $\mathcal{I}_A\subset
K[t]$ und das Minimalpolynom $M_A\in K[t]$.

\textbf{Satz:} Sei $F$ ein Endomorphismus mit Jordanscher Normalform
$J=M_\mathcal{B}(F)$, und seien $\lambda_1, \dots, \lambda_k$ die paarweise
verschiedenen Eigenwerte von $F$. Ferner sie $m_j$ die Größe des größten
Jordan-Blocks zum Eigenwert $\lambda_j$, der in der Jordan-Matrix J vorkommt.
Das Minimalpolynom von $F$ ist dann
\begin{equation*}
	M_F=(t-\lambda_1)^{m_1}\dots(t-\lambda_k)^{m_k}.
\end{equation*}

\textbf{Satz von Cayley-Hamilton:} Sei $V$ ein endlichdimensionaler
$K$-Vektorraum, $F\in End(V)$ und $P_F\in K[t]$ sein charakteristishces
Polynom. Dann ist $P_F(F)=0\in End(V)$. Für die Matrix $A\in M(n\times n;K)$
gilt analog $P_A(A)=0\in M(n\times n;K)$.

\textbf{Korollar}: Das Minimalpolynom $M_F$ teilt das charakteristische Polynom
$P_F$. Insbesondere ist 
\begin{equation*}
	deg\ M_F\leq deg\ P_F=dim\ V.
\end{equation*}

\newpage
\textbf{Definition:} Das kanonische Skalarprodukt (in $\mathbb{R}$) entspricht
einer Abbildung folgender Form 
\begin{equation*}
\langle\cdot,\cdot\rangle: \mathbb{R}^n \times
\mathbb{R}^n\to\mathbb{R}, (x,y)\mapsto\langle x,y\rangle.
\end{equation*}
Sind $x,y\in\mathbb{R}^n$ Spaltenvektoren so ist $\langle x,y\rangle = {^tx}\cdot
y$.

\textbf{Beispiel}: Sei $A,B\in M(n\times n,K)$ und $C=A\cdot B$, sei $a_i$ der
$i$-te Zeilenvektor von $A$, $b_j$ der $j$-te Spaltenvektor von $B$, $c_{i,j}$
ein Eintrag von $C$ entsprechend den Indizes. So ist $c_{i,j}=\langle
a_i,b_j\rangle$

\textbf{Definition: } sei $x,x',y,y'\in\mathbb{R}^n$ und $\lambda\in\mathbb{R}$
so gelten folgende Grundlegende Eigenschaften:
\begin{alignat*}{2}
	1.&\langle x+x',y\rangle = \langle x,y\rangle + \langle
	x',y\rangle,\quad&&
	\langle x,y+y'\rangle = \langle x,y\rangle + \langle x,y'\rangle,\\
	&\langle \lambda x,y\rangle=\lambda\langle x,y\rangle,&&
	\langle x,\lambda y\rangle=\lambda\langle x,y\rangle\\
	2.&\langle x,y\rangle = \langle y,x\rangle\\
	3.&\mathrlap{\langle x,y\rangle \geq0,\text{ wobei }\langle x,x\rangle
	=0\text{ nur für $x=0$ gilt}}
\end{alignat*}

\textbf{Definition:} Aufgrund der Eigenschaften des Skalarprodukts lässt sich
\textit{eine} (induzierte) Norm wie folgt definieren:
\begin{equation*}
||\cdot||:\mathbb{R}^n\to\mathbb{R}_+,\ x\mapsto||x||:=\sqrt{\langle
x,x\rangle}
\end{equation*}

\textbf{Definition:} Die folgende Metrik sei \textit{eine} (induzierte) Metrik:
\begin{equation*}
	d:\mathbb{R}^n\times \mathbb{R}^n\to R_+,\quad d(x,y):=||y-x||.
\end{equation*}

\textbf{Hinweis:} Hier ist besonders betont, dass die Metrik und Norm die
gezeigt wird, eine Metrik/Norm vieler sind und spezifisch diese Norm vom
Skalarprodukt induziert ist und parallel die gegeben Metrik von der vorherigen
Norm induziert ist. Sei das Skalarprodukt entsprechend das kanonische so ist
die induzierte Norm auch bekannt als die Euklidische Norm und die induzierte
Metrik der Euklidische Abstand. Die folgenden Eigenschaften gelten allerdings
für alle Normen und Metriken. Später bei Cauchy-Schwarz sehen wir ein
Verhältnis zwischen Skalarprodukt und deren assozierte Norm was nicht dringend
vorraussetz dass das gegeben Skalarprodukt kanonisch sein muss.

\textbf{Beobachtung:} Die wichtigsten Eigenschaften von Norm und Abstand sind
die folgenden
\begin{align*}
	N1\quad& ||x||=0\iff x=0,\\
	N2\quad& ||\lambda x||=|\lambda|\cdot ||x||,\\
	N3\quad& ||x+y||\leq ||x||+||y||, & \text{Dreiecksungleichung}\\
	D1\quad& d(x,y)=0\iff x=y\\
	D2\quad& d(y,x)=d(x,y), & \text{Symmetrie}\\
	D3\quad& d(x,z)\leq d(x,y)+d(y,z). & \text{Dreiecksungleichung}
\end{align*}

\textbf{Ungleichung von Cauchy-Schwarz:} Für $x,y\in\mathbb{R}^n$ gilt:
\begin{equation*}
	|\langle x,y\rangle|\leq ||x||\cdot ||y||
\end{equation*}
Sind $x,y$ linear abhängig dann gilt:
\begin{equation*}
	|\langle x,y\rangle|=||x||\cdot ||y||
\end{equation*}

\textbf{Definition:} Zwei Vektoren $x,y\in\mathbb{R}^n$ heißen
orthogonal$(\perp)$ wenn $\langle x,y\rangle=0$. Sind $x,y\neq 0$, so gilt
offenbar 
\begin{equation*}
x\perp y\iff\measuredangle(x,y)=\frac{\pi}{2}.
\end{equation*}

\textbf{Definition:} Die Definition für den Winkel kann man auch in folgender
Form schreiben 
\begin{equation*}
\langle x,y\rangle=||x||\cdot||y||\cdot
cos\measuredangle(x,y).
\end{equation*}
Diese Formel erlaubt eine geometrische Erklärung des Skalarprodukts als Produkt
von $||x||$ und $||y||\cdot cos\measuredangle(x,y)$, wobei der zweite Faktor
gleich der senkrechten Projektion des Vektors $y$ auf die durch $x$
aufgespannte Gerade ist.

\textbf{Definition:} Das Vektorprodukt für $n=3$ definiert sich wie folgt:
\begin{alignat*}{2}
	&\mathbb{R}^3\times\mathbb{R}^3\to\mathbb{R}^3,&&(x,y)\mapsto x\times
	y,\\
	&\mathrlap{x\times y:=(x_2y_3-x_3y_2,x_3y_1-x_1y_3,x_1y_2-x_2y_1),}\\
	\text{oder äquivalent: } & \mathrlap{x\times y = det
	\begin{pmatrix}
		e_1 & e_2 & e_3\\
		x_1 & x_2 & x_3\\
		y_1 & y_2 & y_3
	\end{pmatrix}}
\end{alignat*}
Für das Vektorprodukt, $x,x',y,y'\in\mathbb{R}^3$ und $\lambda\in\mathbb{R}$
gelten folgende grundlegende Rechenregeln:
\begin{alignat*}{2}
	1.\quad&(x+x')\times y=x\times y+x'\times y \quad&& x\times(y+y')=x\times
	y+x\times y',\\
	& \lambda x\times y=\lambda(x\times y), && x\times \lambda
	y=\lambda(x\times y),\\
	2.\quad&y\times x=-x\times y,  && x\times x=0,\\
	3.\quad&\mathrlap{x\times y= 0\iff x,y\text{ linear unabhängig.}}
\end{alignat*}

\textbf{Bemerkung:} Für $x,y,z\in\mathbb{R}^3$ gilt:
\begin{alignat*}{2}
	a)\quad&\langle x\times y, z\rangle=det
	\begin{pmatrix}
		x_1&x_2&x_3\\
		y_1&y_2&y_3\\
		z_1&z_2&z_3
	\end{pmatrix},\quad&&
	\langle x\times y,x\rangle = \langle x\times y, y\rangle =0\\
	b)\quad&||x\times y||^2=||x||^2||y||^2-\langle x,y\rangle^2, && 
	||x\times y||=||x||\cdot||y||\cdot sin\measuredangle(x,y).
\end{alignat*}

Das Produkt $\langle x\times y,z\rangle$ wird auch Spatprodukt genannt.

\textbf{Grassmann-Identität:}
\begin{equation*}
	x\times (y\times z) = \langle x,z\rangle y - \langle x,y\rangle z
\end{equation*}

\textbf{Jacobi-Identität:}
\begin{equation*}
	x\times(y\times z)+y\times(z\times x)+z\times(x\times y)=0
\end{equation*}

\textbf{Definition:} Zu einer komplexen Zahl $z=x+iy$ ist definiert
$\bar{z}=x-iy$ und $|z|=\sqrt{z\bar{z}} =\sqrt{x^2+y^2}$. 

Die Zahl $z\in\mathbb{C}$ entspricht der Vektor $(x,y)\in\mathbb{R}^2$, und mit
der euklidischen Norm gilt dann $|z|=||(x,y)||$.

Für die Geometrie des $\mathbb{C}^n$ ist die folgende Abbildung entsprechend
dem kanonischen Skalarprodukt:
\begin{alignat*}{2}
	&\mathbb{C}^2\times \mathbb{C}^2\to\mathbb{C}\quad, &&
	(z,w)\mapsto\langle z,w\rangle\rangle_c\\
	&\langle z,w\rangle_c:=z_1\bar{w}_1+\dots+z_n\bar{w}_n
\end{alignat*}
Für das kanonische Skalarprodukt in $\mathbb{C}$ mit $z,z',w,w'\in\mathbb{C}^n$
und $\lambda\in\mathbb{C}$ gelten folgende Rechenreglen:
\begin{alignat*}{2}
	1.&\langle z+z',w\rangle = \langle z,w\rangle + \langle
	z',w\rangle,\quad&&
	\langle z,w+w'\rangle = \langle z,w\rangle + \langle z,w'\rangle,\\
	&\langle \lambda z,w\rangle=\lambda\langle z,w\rangle,&&
	\langle z,\lambda w\rangle=\bar\lambda\langle z,w\rangle\\
	2.&\langle z,w\rangle = \overline{\langle w,z\rangle}\\
	3.&\mathrlap{\langle z,z\rangle \in\mathbb{R}_+,\text{ und }\langle
	z,z\rangle
	=0\iff z=0}
\end{alignat*}
\textbf{Hinweis:} Der Index $c$ wird vorübergehend angehängt um auf den
Unterschied zum Skalarprodukt im $\mathbb{R}^n$ hinzuweisen. Zu bemerken ist,
dass das hier definierte kanonische Skalarprodukt angewendet auf reele Zahlen
dem kanonischen Produkt entspricht was zuvor definiert wurde. Eben hier besteht
oft Verwirrung, den in $\mathbb{R}$ behandelt man das Skalarprodukt als
bilinear, also linear in beiden Argumenten, allerdings ist es im allgemeinen
nicht bilinear!

Weiterhin können unterschiedliche Quellen verschiedene Konventionen haben was
es Reihenfolge betrifft. Beispielsweise wird in mindestens
\href{https://de.wikipedia.org/wiki/Gram-Schmidtsches_Orthogonalisierungsverfahren
}{einem Wikipedia Artikel} angemerkt, dass das Skalarprodukt benutzt wird
welches semilinear im ersten und linear im zweiten ist.

\textbf{Bilinearform:} 
\begin{alignat*}{2}
	&\mathrlap{s:V\times V\to K,\quad(v,w)\mapsto s(v,w),}\\
	B1\quad&s(v+v',w)=s(v,w)+s(v',w),\quad && s(\lambda v,w)=\lambda
	s(v,w),\\
	B2\quad&s(v,w+w')=s(v,w)+s(v,w'),&& s(v,\lambda w)=\lambda s(v,w).
\end{alignat*}
Weiterhin kann man Bilinearform wie folgt kategorisieren entsprechend der
jeweiligen Eigenschaft
\begin{alignat*}{2}
	&\text{symmetrisch:  } &&s(v,w) = s(w,v)\\
	&\text{alternierend:  } &&s(v,w) = -s(w,v)
\end{alignat*}

\textbf{Bemerkung:} Sei $S$ eine Bilinearform auf $V$ mit Basis
$\mathcal{B},\Phi_\mathcal{B}:K^n\to V$ das zugehörige Koordinatensystem. Wir
betrachten die Matrix $A=M_\mathcal{B}(s)$ und für $v,w\in V$ die Koordinaten
$x=\Phi_\mathcal{B}^{-1}(v),y=\Phi_\mathcal{B}^{-1}(w)$. Dann ist
\begin{equation*}
	s(v,w)={^tx}Ay=(x_1,\dots,x_n)
	\begin{pmatrix}
		a_{1,1}&\dots &a_{1,n}\\
		\vdots & & \vdots\\
		a_{n,1}&\dots &a_{n,n}\
	\end{pmatrix}
	\begin{pmatrix}
		y_1\\ \vdots\\ y_n
	\end{pmatrix}
\end{equation*}
Weiterhin hat $A$ folgende Form
\begin{equation*}
	A=M_\mathcal{B}(s)=(s(v_i,v_j))_{i,j}\in M(n\times n;K)
\end{equation*}

\textbf{Satz:} Sei $V$ ein $K$-Vektorraum mit $n=dim\ V<\infty$ und
$\mathcal{B}$ eine Basis. Dann ist die Abbildung $s\mapsto M_\mathcal{B}(s)$
von den Bilinearformen auf $V$ in $M(n\times n; K)$ bijektiv, und $s$ ist genau
dann symmetrisch, wenn $M_\mathcal{B}(s)$ symmetrisch ist.

\textbf{Transformationsformel:} Sei $V$ ein endlichdimensionaler Vektorraum mit
Basen $\mathcal{A,B}$ und sei $T^\mathcal{B}_\mathcal{A}$ die entsprechende
Transformationsmatrix der Koordinaten. Für jede Bilinearform $s$ auf $V$ gilt
dann:
\begin{equation*}
	M_\mathcal{B}(s)={^tT^\mathcal{B}_\mathcal{A}}\cdot
	M_\mathcal{A}(s)T^\mathcal{B}_\mathcal{A}
\end{equation*}

\textbf{Hinweis:} Hier sei ein struktureller Unterschied zu beobachten denn es gilt:
\begin{alignat*}{3}
	&B= {^tT}^{-1}AT^{-1} &&= {^tS}AS && \quad\text{ für Bilineaformen}\\
	&B=TAT^{-1} &&=S^{-1}AS && \quad\text{ für Endomorphismen.}
\end{alignat*}

\textbf{Definition:} Sei $s$ eine symmetrische Bilinearform $s:V\times V \to K$
dann ist $q$ die assoziierte quadratische Form mit
\begin{equation*}
	q:V\to K,\quad v\mapsto q(v) :=s(v,v)
\end{equation*}
Wobei folgende Eigenschaften gelten:
\begin{alignat*}{1}
	&q(\lambda v) = \lambda^2q(v)\\
	&s(v,w)=\frac{1}{2}(q(v+w)-q(v)-q(w))
\end{alignat*}

\textbf{Diagonalisierungssatz:} Sei $K$ ein Körper mit $char(K)\neq 2$, $V$ ein
endlichdimensionaler $K$-Vektorraum und $s:V\times V\to K$ eine symmetrische
Bilinearform. Dann gibt es eine Basis $\mathcal{B}=(v_1,\dots v_n)$ mit
$s(v_i,v_j)=0$ für $i\neq j$. Ist $q:V\to K$ die zugehörige quadratische Form
und $\alpha_i:=q(v_i)=s(v_i,v_i)$, so folgt
\begin{equation*}
	q(v)=\alpha_1x_1^2+\dots + a_nx_n^2\quad \text{für}
	v=x_1v_1+\dots+x_nv_n\in V
\end{equation*}
Ist $rang(s)=r$, so kann man die Reihenfolge derart wählen dass
\begin{equation*}
	\alpha_1\neq 0, \dots,\alpha_r\neq 0\text{ und
	}\alpha_{r+1}=\dots=\alpha_n=0.
\end{equation*}

\textbf{Korollar:} Zu einer symmetrischen Matrix $A\in M(n\times n;K)$ mit
$char(k)\neq 2$ gibt es ein $S\in GL(n;K)$, sodass
\begin{equation*}
	{^tS}AS=\begin{pmatrix}
		\alpha_1 & & 0\\
		& \ddots & \\
		0 & & \alpha_n
	\end{pmatrix}
\end{equation*}

\textbf{Lemma:} Für jede symmetrische Bilinearform $s$ und jede Basis
$\mathcal{B}$ von $V$ ist $rang(s)=rang\ M_\mathcal{B}(s)$.

\textbf{Trägheitsgesetz von Sylvester:} Hat man für eine quadratische Form $q$
auf einem $\mathbb{R}$-Vektorraum $V$ zwei Zerlegungen $V=V_+\oplus V_-\oplus
V_0 = V'_+\oplus V'_-\oplus V_0$ mit 
\begin{alignat*}{1}
	& q(v)>0\text{ für } 0\neq v\in V_+\text{ und } 0\neq v\in V_+'\text{
	sowie}\\
	& q(v)<0\text{ für } 0\neq v\in V_-\text{ und } 0\neq v\in V_-'
\end{alignat*}
so folgt
\begin{equation*}
	dim\ V_+'=dim\ V_+\text{  und  }dim\ V'_-=dim_-.
\end{equation*}

	s(v,w),\\
	\overline{B2}\quad&s(v,w+w')=s(v,w)+s(v,w'),&&\quad s(v,\lambda
	w)=\bar\lambda s(v,w),\\
\end{alignat*}
weiterhin gilt eine Sesqiulinearform als hermitesch wenn gilt:
\begin{equation*}
	H\quad s(v,w)=\overline{s(w,v)}
\end{equation*}

\textbf{Definition:} Sei $\mathbb{K}\in\{\mathbb{R,C}\}$, und $V$ ein
$\mathbb{K}$-Vektorraum. Eine symmetrische Bilinearform/hermitesche
Sesquilnearform $s:V\times V\to K$ ist dann positiv definit, wenn $s(v,v)>0$
für jedes $v\in V,v\neq 0$. 

Sei $A\in M(n\times n,\mathbb{K})$ symmetrisch/hermitesch so heißt $A$ positiv
definit, wenn ${^tx}A\bar x>0$ für jeden Spaltenvektor $x\neq0$ aus $K^n$.

\textbf{Hinweis:} Aus dieser Definition lässt sich positiv semidefinit, negativ
definit, negativ semidefinit und indefinit leicht definieren, was hier nicht
weiter aufgeführt wird.

\textbf{Hauptminoren-Kriterium für Definitheit:} Sei $A\in M(n\times
n;\mathbb{R})$ eine symmetrische Matrix. Dann gilt:
\begin{equation*}
	A\text{ ist positiv definit} \iff \forall k=1,\dots,n: det\ A_k>0 
\end{equation*}

\textbf{Hinweis:} informell ist hier $A_k$ gemeint als die oberste linkeste
Matrix mit größe $k\times k$ von $A$.

\textbf{Satz:} Für eine symmetrische bzw. hermitesche Matrix $A\in M(n\times n;
\mathbb{K})$ sind folgende Bedingungen gleichwertig:
\begin{itemize}
\item[i)] $A$ ist positiv definit.
\item[ii)] Alle Eigenwerte $\lambda_1,\dots, \lambda_n\in\mathbb{R}$ von $A$
	sind positiv.
\end{itemize}

\textbf{Definition:} Gegeben sei ein Körper $\mathbb{K}\in\{\mathbb{R,C}\}$ und
$\mathbb{K}$-Vektorraum $V$. Ein Skalarprodukt ist eine positiv definite
symmetrische Bilinearform für $\mathbb{K}=\mathbb{R}$ bzw. eine positiv
definite hermitesche Sesquilinearform für $\mathbb{K}=\mathbb{C}$. 

Ein $\mathbb{R}$-Vektorraum/$\mathbb{C}$-Vektorraum mit einem Skalarprodukt
heißt euklidisch/unitär.

\textbf{Hinweis:} Wir schreiben dann meistens $\langle v,w\rangle$ statt
$s(v,w)$.

\textbf{Beobachtung:} Die Abbildung 
\begin{equation*}
||\cdot||:V\to\mathbb{R},\quad ||v||=\sqrt{\langle v,v\rangle}
\end{equation*}
ist eine Norm und ferner $d(v,w):=||w-v||$ eine (induzierte) Metrik.

\textbf{Ungleichung von Cauchy-Schwarz:} Ist $V$ ein euklidischer bzw. unitärer
Vektorraum und $||\cdot||$ induziert aus $\langle\cdot,\cdot\rangle$ so gilt
für alle $v,w\in V$
\begin{equation*}
	|\langle v,w\rangle|\leq||v||\cdot||w||,
\end{equation*}
und die Gleichheit gilt genau dann, wenn $v$ und $w$ linear abhängig sind.

\newpage
\textbf{Definition:} Sie $V$ ein euklidischer bzw. unitärer Vektorraum

a) Zwei Vektoren $v,w\in V$ heißen orthogonal ($v\perp w$) genau dann wenn
$\langle v,w\rangle=0$.

b) Zwei Untervektorräume $U,W\subset V$ heißen orthogonal ($U\perp W$) genau
dann wenn $\forall u\in U,w\in W:\ u\perp w$.

c) Ist $U\subset V$ ein Untervektorraum, so ist sein orthogonales Komplement
wieder ein Untervektorraum von $V$ und definiert als $U^\perp:=\{v\in V:v\perp
u,\quad \forall u\in U\}$.

d) Eine Familie $(v_1,\dots,v_n)$ in $V$ heißt orthogonal, wenn $v_i\perp v_j$
für alle $i\neq j$.

Sie heißt orthonormal, falls zusätzlich $||v_i||=1$,

und Orthonormalbasis, falls sie auch eine Basis ist, d.h. eine Basis
mit $\langle v_i,v_j\rangle=\delta_{ij}$.

e) Ist $V=V_1\oplus\dots\oplus V_k$, so heißt die direkte Summe orthogonal
($V=V_1\operp\dots\operp V_k$) falls $V_i\perp V_j$ für alle $i\neq j$

\textbf{Orthonormalisierungssatz:} Sei $V$ ein endlichdimensionaler
euklidischer bzw. unitärer Vektorraum und $W\subset V$ ein Untervektorraum mit
Orthonormalbasis $(w_1,\dots, w_m)$. Dann gibt es eine Ergänzung zu einer
Orthonormalbasis $(w_1,\dots,w_m, w_{m+1},\dots,w_n)$ von $V$.

\textbf{Korollar 1:} Jeder endlichdimensionale euklidische bzw. unitäre
Vektorraum besitzt eine Orthonormalbasis.

\textbf{Korollar 2:} Ist $W$ Untervektorraum eines endlichdimensionalen
euklidischen bzw. unitären Vektorraums $V$, so gilt $V=W\operp W^\perp$ und
$dim\ V=dim\ W+dim\ W^\perp$.

\textbf{Gram-Schmidt-Orthogonalisierungsverfahren:} Sei gegeben eine Menge an
linear unabhängigen Vektoren $w_1,\dots,w_n$, so berechnet dieser Algorithmus
eine Orthonormalbasis die den selben Untervektorraum erzeugt.

\begin{enumerate}
	\item $v_1=\frac{w_1}{||w_1||}$
	\item $v'_i = w_i - \sum^{n-1} \langle w_i,v_j\rangle v_j$ 
	\item $v_i = \dfrac{v'_i}{||v'_i||}$
	\item $(v_1,\dots, v_n)$ ist dann eine Orthonormalbasis
\end{enumerate}

\textbf{Hinweis:} In der Vorlesung ist dieses Verfahren nicht explizit
beschrieben und im der Literatur nur impliziert durch den Beweis des
Orthonormalisierungssatzes. In weiteren Quellen lassen sich zwei Varianten
finden, jeweils Orthogonalisierung und Orthonormalisierung. Zusätzlich ist
verwirrend, dass Gram-Schmidtsche Orthogonalisierungsverfahren beschrieben wird
als ein Verfahren zu finden einer Orthonormalbasis. Je nach einer oder anderen
Variante normalisiert man an verschiedenen Stellen und erhält möglicherweise
als Ergebnis nicht normalisierte Vektoren im gegensatz zu dem hier aufgeführtem
Verfahren.

\textbf{Definition:} Die Gramsche Determinante sei definiert als
\begin{equation*}
	G(v_1,\dots,v_m)=det
	\begin{pmatrix}
		\langle v_1, v_1\rangle&\dots & \langle v_1, v_m\rangle \\
		\vdots & \ddots & \vdots\\
		\langle v_m, v_1\rangle&\dots &\langle v_m, v_m\rangle
	\end{pmatrix}
\end{equation*}

\textbf{Lemma:} Für $m\leq n$ und $v_1,\dots,v_m\in\mathbb{R}^n$ ist
$G(v_1,\dots,v_m)\geq 0$ und
\begin{equation*}
 G(v_1,\dots,v_m)=0\iff(v_1,\dots,v_m) \text{  linear abhängig.}
\end{equation*}

\textbf{Satz:} Für $m\leq n$ und linear unabhängige Vektoren $v_1,\dots,
v_m\in\mathbb{R}^n$ ist
\begin{equation*}
	I_m(P(v_1,\dots, v_m))=\sqrt{G(v_1,\dots, v_m)}
\end{equation*}

\textbf{Ungleichung von Hadamard:} Seien $v_1,\dots, v_m$ beliebige Vektoren im
$\mathbb{R}^n$ mit $m\leq n$. Dann ist 
\begin{equation*}
	I_m(P(v_1,\dots, v_m))\leq ||v_1||\cdot\dots\cdot||v_m||,
\end{equation*}
und Gleichheit besteht genau dann, wenn die $v_i$ paarweise orthogonal sind,
d.h. wenn $P(v_1,\dots,v_m)$ ein Quader ist.

\textbf{Definition:} Sei $V$ ein euklidischer bzw. unitärer Vektorraum und $F$
ein Endomorphismus von $V$. Dann heißt $F$ orthogonal bzw. unitär wenn
\begin{equation*}
	\langle F(v),F(w)\rangle = \langle v,w\rangle
\end{equation*}

\textbf{Bemerkung:} ein orthogonales bzw. unitäres $F\in End(V)$ hat folgende
weiter Eigenschaft:

\begin{itemize}
\item[a)] $||F(v)||=||v||$ für alle $v\in V$.
\item[b)] $v\perp w\implies F(v)\perp F(w)$.
\item[c)] $F$ ist Isomorphismus und $F^{-1}$ ist orthogonal bzw. unitär
\item[d)] Ist $\lambda\in\mathbb{K}$ Eigenwert von $F$, so ist $|\lambda|=1$.
\end{itemize}

\textbf{Lemma:} Ist $F\in End(V)$ mit $||F(v)||=||v||$ für alle $v\in V$, so
ist $F$ orthogonal bzw. unitär.

\textbf{Definition:} Eine Matrix $A\in GL(n;\mathbb{R}$ heißt orthogonal, falls
$A^{-1}={^tA}$, und entsprechend heißt $A\in GL(n;\mathbb{C})$ unitär, falls
$A^{-1}=\overline{^tA}$

\newpage
\textbf{Bemerkung:} Für $A\in M(n\times n;\mathbb{K})$ sind folgende
Bedingungen gleichwertig:
\begin{itemize}
\item[i)] $A$ ist orthogonal bzw. unitär.
\item[ii)] Die Spalten von $A$ sind eine Orthonormalbasis von $\mathbb{K}^n$.
\item[iii)] Die Zeilen von $A$ sind eine Orthonormalbasis von $\mathbb{K}^n$.
\end{itemize}

\textbf{Definition:}
\begin{alignat*}{2}
	&O(n):=\{A\in GL(n;\mathbb{R}):A^{-1}={^tA}\},\quad && \text{(orthogonale
	Gruppe)}\\
	&SO(n):=\{A\in O(n):det\ A=1\},\quad && \text{(spezielle orthogonale
	Gruppe)}\\
	&U(n):=\{A\in GL(n;\mathbb{C}):A^{-1}=\overline{^tA}\},\quad &&
	\text{(unitäre Gruppe)}\\
	&SU(n):=\{A\in U(n):det\ A=1\},\quad && \text{(spezielle unitäre
	Gruppe)}\\
\end{alignat*}

\textbf{Satz:} Sei $V$ ein euklidischer bzw. unitärer Vektorraum mit einer
Orthonormalbasis $\mathcal{B}$ und $F$ ein Endomorphismus von $V$. Dann gilt
\begin{equation*}
	F\text{ orthogonal (bzw. unitär)}\iff M_\mathcal{B}(F)\text{ orthogonal
	(bzw. unitär).}
\end{equation*}

\textbf{Diagonalisierungssatz:} Jeder unitäre Endomorphismus $F$ eines unitären
Vektorraums besitzt eine Orthonormalbasis aus Eigenvektoren von $F$.
Insbesondere ist er diagonalisierbar.

\textbf{Korollar:} Zu $A\in U(n)$ gibt es ein $S\in U(n)$ mit

\end{equation*}
wobei $\lambda_i\in\mathbb{C}$ mit $|\lambda_i|=1$ für $i=1,\dots,n$.

\textbf{Satz:} Ist $F$ ein orthogonaler Endomorphismus eines euklidischen

wobei für $j=1,\dots,k$

\textbf{Lemma:} Sei $V$ ein endlichdimensionaler euklidischer bzw. unitärer
$\mathbb{K}$-Vektorraum und $F:V\to V$ ein Endomorphismus. Dann gibt es genau
einen Endomorphismus
\begin{equation*}
	F^{ad}:V\to V\quad\text{mit}\quad \langle F(v),w\rangle=\langle
	v,F^{ad}(w)\rangle\quad \text{für alle } v,w\in V.
\end{equation*}
Ist $\mathcal{B}$ eine Orthonormalbasis von $V$ und $A:=M_\mathcal{B}(F)$, so
ist $M_\mathcal{B}(F^{ad})={^t\bar A}$. Dabei heißt $F^{ad}$ der zu $F$
adjungierte Endomorphismus.

\textbf{Definition:} Ist $V$ ein euklidischer oder unitärer Vektorraum, so
heißt ein Endomorphismus $F$ von $V$ \textbf{normal}, wenn
\begin{equation*}
	F\circ F^{ad}=F^{ad}\circ F
\end{equation*}
Ist $A=M_\mathcal{B}(F)$ mit einer Orthonormalbasis $\mathcal{B}$ von $V$, so
bedeutet das
\begin{equation*}
	A\cdot {^t\bar A}= {^t\bar A}\cdot A.
\end{equation*}

\textbf{Beispiel:}
\begin{itemize}
\item alle unitären Endomorphismen: hier gilt ${^t\bar A=A^{-1}}$, also
	$A\cdot{^t\bar A}= E = {^t\bar A}\cdot A$
\item alle hermiteschen Endomorphismen: hier gilt ${^t\bar A=A^{-1}}$, also
	$A\cdot{^t\bar A}= A^2 = {^t\bar A}\cdot A$
\end{itemize}

\textbf{Lemma:} Sei $V$ ein euklidischer bzw. unitärer Vektorraum und $F:V\to
V$ ein Endomorphismus. Dann gilt
\begin{itemize}
\item[1)] $F$ normal $\iff\ \langle F^{ad}(v),F^{ad}(w)\rangle=\langle
	F(v),F(w)\rangle$ für alle $v,w\in V$.
\item[2)] $F$ normal $\implies\forall v\in V:||F^{ad}(v)||=||F(v)||$ und $Ker\
	F^{ad}=Ker\ F$.
\item[3)] Ist $F$ normal, so ist $G:=F-\lambda I_V$ für alle
	$\lambda\in\mathbb{K}$ normal.
\item[4)] Für alle $v\in V$ und $\lambda\in\mathbb{K}$ gilt $F(v)=\lambda v\iff
	F^{ad}(v)=\bar\lambda v$.
\end{itemize}

\textbf{Diagonalisierungssatz:} Für einen Endomorphismus $F$ eines unitären
Vektorraums $V$ sind folgende Eigenschaften gleichwertig:
\begin{itemize}
\item[i)] $F$ ist normal
\item[ii)] Es gibt eine Orthonormalbasis $\mathcal{B}$ von $V$ bestehend aus
	Eigenvektoren von $F$.
\end{itemize}

\textbf{Definition:} Ein Endomorphismus $F$ eines endlichdimensionalen
euklidischen bzw. unitären Vektorraums $V$ heißt \textbf{selbstadjungiert},
wenn $F=F^{ad}$, d.h.
\begin{equation*}
	\langle F(v),w\rangle=\langle v, F(w)\rangle\quad\text{für alle }
	v,w\in V
\end{equation*}

\textbf{Bemerkung:} Ist $F\in End(V)$ und $\mathcal{B}$ eine Orthonormalbasis
von $V$, so gilt $F$ selbstadjungiert $\iff$ $M_\mathcal{B}(F)$ ist symmetrisch
bzw. hermitesch, d.h. $A={^t\bar A}$.

\textbf{Lemma:} Sei $V$ ein euklidischer bzw. unitärer $\mathbb{K}$-Vektorraum
und $F:V\to V$ selbstadjungiert. Dann gilt:
\begin{itemize}
\item[i)] Alle Eigenwerte von $F$ sind reell.
\item[ii)] $P_F$ hat auch für $\mathbb{K=C}$ reelle Koeffizienten.
\item[iii)] $P_F$ zerfällt in Linearfaktoren in $\mathbb{R}[t]$.
\end{itemize}

\textbf{Diagonalisierungssatz:} Ist $F$ ein selbstadjungierter Endomorphismus
auf einem euklidischen bzw. unitären Vektorraum $V$, so gibt es eine
Orthonormalbasis von $V$, die aus Eigenvektoren von $F$ zu reellen Eigenwerten
besteht.

\textbf{Korollar:} Ist $A\in M(n\times n;\mathbb{K})$ eine symmetrische bzw.
hermitesche Matrix so gibt es eine orthogonale bzw. unitäre Matrix $S$, sodass
\begin{equation*}
	^t\bar SAS=
\begin{pmatrix}
	\lambda_1&&0\\
	&\ddots&\\
	0&&\lambda_n\\
\end{pmatrix}
	\quad\text{mit } \lambda_1,\dots,\lambda_n\in\mathbb{R}.
\end{equation*}

\textbf{Informell:} 
\begin{itemize}
\item[] unitär $\subset$ normal $\subset$ Endomorphismen
\item[] selbstadjungiert $\subset$ normal $\subset$ Endomorphismen
\item[] unitär $\cap$ selbstadjungiert $\neq \varnothing$
\end{itemize}

\textbf{Korollar:} Sind $\lambda_1,\dots,\lambda_k$ die verschiedenen Eigenwerte
eines selbstadjungierten oder unitären Endomorphismus $F$ von $V$, so ist
\begin{equation*}
	V=Eig(F;\lambda_1)\operp\dots\operp Eig(F;\lambda_k).
\end{equation*}

\textbf{Satz:} Für eine symmetrische bzw. hermitesche Matrix $A\in M(n\times
n;\mathbb{K})$ sind folgende Bedingungen gleichwertig:
\begin{itemize}
\item[i)] $A$ ist positiv definit.
\item[ii)] Alle Eigenwerte $\lambda_1,\dots,\lambda_n\in\mathbb{R}$ von $A$
	sind positiv.
\end{itemize}

\textbf{Korollar:} Ist $A\in M(n\times n;\mathbb{K}$ symmetrisch bzw.
hermitesch und
\begin{equation*}
P_A=(-1)^nt^n+\alpha_{n-1}t^{n-1}+\dots+\alpha_1t+\alpha_0\in\mathbb{R}[t]
\end{equation*}
das charakteristische Polynom, so gilt:
\begin{equation*}
	A\text{ positiv definit}  \iff \left\{ 
\begin{aligned}
    \alpha_j>0& \quad \text{für $j$ gerade} \\
    \alpha_j<0& \quad \text{für $j$ ungerade.}
\end{aligned}
\right.
\end{equation*}
Insbesondere muss $\alpha_0=det\ A$ positiv sein.

\textbf{Beobachtung:} Zusammengefasst sind folgende Kriterien equivalent
\begin{itemize}
\item[i)] $A\in M(n\times n,\mathbb{K})$ ist positiv definit
\item[ii)] ${^tx}A\bar x>0\quad\forall x\in\mathbb{K}^n\backslash\{0\}$.
\item[iii)] $det\ A_k>0$ für $k=1,\dots,n$ wobei $A_k$ obere $k\times
	k$-Matrix.
\item[iiii)] Alle Eigenwerte von $A$ sind positiv.
\item[iiiii)] Für $P_A$ gilt
	$A\text{ positiv definit}  \iff \left\{ 
\begin{aligned}
    \alpha_j>0& \quad \text{für $j$ gerade} \\
    \alpha_j<0& \quad \text{für $j$ ungerade.}
\end{aligned}
\right.$
\end{itemize}
